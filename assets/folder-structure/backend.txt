    controllers/
      alertController.ts       — Alert CRUD + push notification broadcast
      authController.ts        — Register, Login, Verify, Profile CRUD, Role mgmt
      contactController.ts     — Emergency contact CRUD by department/location

      reportModel.ts            — Report schema (type, description, image, location, status)
      safeZoneModel.ts          — SafeZone schema (name, type, location, address, phone, isActive)
      userModel.ts              — User schema (username, password, phone, role, location, etc.)
